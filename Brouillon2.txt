void	print_stack(t_stack *stack)
{
	t_stack		*tmp;

	if (!stack)
		return ;
	tmp = stack;
	while (tmp)
	{
		ft_printf("%d ", *(int *)tmp->content);
		tmp = tmp->next;
	}
	ft_printf("\n");
}
